**Goal:** Make `back-ui-821-other-keep` a stable base that ingests non-lobehub architectural improvements from main without breaking the lobehub 8.21 frontend contract. Deliver ADR-0103, a 816-row classification of main's commits, 47 cluster end-state cards, a hard-lock guard script, and a port-execution pattern.

**Architecture:** Lock surface is enforced by `scripts/check_locked_surface.py` (parses ADR-0103's hard-lock table); soft-lock paths emit warnings unless the commit body mentions `wire-shape preserved`. End-state cards are generated by `scripts/port_endstate.py` (`git diff bae32d8c..origin/main -- <paths>` per cluster, summarized). Ports are applied per-cluster as one commit per cluster; one extra step per soft-locked cluster is a manual wire-shape review.

**Tech Stack:** Python 3.12+ stdlib only for the new scripts (no new deps); `git`, `uv run`, existing test infra; markdown for human-facing artifacts.

**Spec:** `docs/superpowers/specs/2026-08-30-back-ui-821-port-policy-design.md`

## Global Constraints

These come straight from the spec and apply to every task below:

- **Hard-lock**: `deploy/lobehub/*` (entire tree including `patches/`); `lobehub-ui/` (out of repo). `check_locked_surface.py` MUST fail any diff that touches these paths.

- **Branch**: all commits land on `back-ui-821-other-keep`; no merges from `origin/main`.
- **Pre-commit on every Python change**: ruff check --fix, ruff format, lint-imports, mypy lca, check_no_any, check_no_bare_strings, check_assembly_purity, check_protocol_impl, check_plugin_typing, vulture (≥80 confidence).

- **Commit author**: `LCA Agent <lca-agent@local>` (matches existing branch convention).
- **No new TODO/FIXME placeholders** in any new file.
- **Pre-commit skip**: use `git commit --no-verify` only when a guard script is being added in the same commit (see Task 3 step 5).

```python
#!/usr/bin/env python3
"""Pre-commit guard: enforce ADR-0103's hard-lock surface.

Reads the hard-lock paths from ADR-0103 (Section 1: Hard-lock) and
checks `git diff <base>..HEAD` for any non-whitespace change to those
paths. Exits 1 if violated, 0 otherwise. Prints violations to stderr.

Soft-lock paths emit a warning unless the commit body of HEAD contains
the phrase "wire-shape preserved".
"""

from __future__ import annotations

import argparse
import re
import subprocess
import sys
from pathlib import Path

_ROOT = Path(__file__).resolve().parent.parent

# Path prefixes parsed from the ADR's "Hard-lock" subsection.
_HARD_LOCK_HEADER = "### 1. Hard-lock"
_SOFT_LOCK_HEADER = "### 2. Soft-lock"


def _parse_lock_paths(adr_path: Path, header: str) -> list[str]:
    """Extract `- path` entries from the subsection under `header`."""
    if not adr_path.exists():
        return []
    text = adr_path.read_text(encoding="utf-8")
    in_section = False
    paths: list[str] = []
    for line in text.splitlines():
        stripped = line.strip()
        if stripped.startswith("###"):
            if in_section:
                break
            if stripped == header:
                in_section = True
            continue
        if in_section and stripped.startswith("- `") and "`" in stripped[3:]:
            # Extract the path inside backticks
            path = stripped.split("`")[1]
            paths.append(path.rstrip("/"))
    return paths


def _diff_names(base: str) -> list[str]:
    """Return changed file paths from `git diff <base>..HEAD`."""
    proc = subprocess.run(
        ["git", "diff", "--name-only", f"{base}..HEAD"],
        capture_output=True,
        text=True,
        check=False,
        cwd=_ROOT,
    )
    if proc.returncode != 0:
        return []
    return [line.strip() for line in proc.stdout.splitlines() if line.strip()]


def _head_commit_message() -> str:
    proc = subprocess.run(
        ["git", "log", "-1", "--format=%B"],
        capture_output=True,
        text=True,
        check=False,
        cwd=_ROOT,
    )
    return proc.stdout


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--adr", default="docs/adr/0103-locked-surface-and-port-policy.md")
    parser.add_argument("--base", default="HEAD")
    parser.add_argument("--fake-diff", action="append", default=[],
                        help="Testing hook: pretend these paths changed.")
    args = parser.parse_args()

    adr_path = _ROOT / args.adr
    hard = _parse_lock_paths(adr_path, _HARD_LOCK_HEADER)
    soft = _parse_lock_paths(adr_path, _SOFT_LOCK_HEADER)

    changed = list(_diff_names(args.base))
    if args.fake_diff:
        # For tests; each entry is "path: body" but we only need the path
        changed = list({*changed, *(entry.split(":", 1)[0] for entry in args.fake_diff)})

    violations: list[str] = []
    warnings: list[str] = []

    for path in changed:
        for lock in hard:
            if path.startswith(lock + "/") or path == lock:
                violations.append(f"  HARD-LOCK violation: {path} (locked by {lock})")
        for lock in soft:
            if path.startswith(lock + "/") or path == lock:
                msg = _head_commit_message()
                if "wire-shape preserved" not in msg:
                    warnings.append(
                        f"  SOFT-LOCK warning: {path} — commit body must mention "
                        "'wire-shape preserved'"
                    )

    for v in violations:
        print(v, file=sys.stderr)
    for w in warnings:
        print(w, file=sys.stderr)

    return 1 if violations else 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Auto-classify main's commits since merge-base into clusters + lanes.

Reads `git log bae32d8c..origin/main`, applies path + commit-message

def classify_commit(sha: str, subject: str, files: list[str]) -> Classification:
    """Classify a single commit into (cluster, lane, notes)."""
    # Lane-A test paths take priority
    for f in files:
        if f.startswith("tests/") and (
            "lca/contracts" in subject or "lca/harness" in subject or "adr" in subject
        ):
            return Classification(sha, subject, "C44", "A", "tests for contracts/harness/docs")

    # Chore / cleanup / format
    if subject.startswith("chore:") or subject.startswith("style:") or subject.startswith("docs:"):
        return Classification(sha, subject, "skip-chore", "C", "chore/style/docs")

    # DSH-only
    if "dsh" in subject.lower() and "lca/layer0_infra/dsh" not in str(files):
        return Classification(sha, subject, "skip-dsh", "C", "DSH-only change")

    # YAGNI cleanup (kill <alias>, delete <file>) — skip unless branch has the alias
    if subject.startswith("refactor:") and ("remove" in subject or "kill" in subject or "delete" in subject):
        return Classification(sha, subject, "skip-yagni", "C", "YAGNI cleanup")

    # Default path-based classification
    for f in files:
        for prefix, cluster, lane in CLUSTER_MAP:
            if f.startswith(prefix):
                notes = f"touches {prefix}"
                return Classification(sha, subject, cluster, lane, notes)

    # No path match — fall back to commit message keyword
    if "adr" in subject.lower():
        return Classification(sha, subject, "C7", "A", "ADR-related (no path)")
    if "harness" in subject.lower():
        return Classification(sha, subject, "C9", "A", "harness-related (no path)")

    return Classification(sha, subject, "unclassified", "B", "needs manual review")


def _git_log(base: str, head: str) -> list[tuple[str, str, list[str]]]:
    """Yield (sha, subject, files) tuples for each commit in `base..head`."""
    fmt = "%H%n%s%n--FILES--%n"
    log_proc = subprocess.run(
        ["git", "log", f"{base}..{head}", f"--pretty=format:{fmt}"],
        capture_output=True,
        text=True,
        check=False,
        cwd=_ROOT,
    )
    if log_proc.returncode != 0:
        return []

    # Parse out subjects
    entries: list[tuple[str, list[str]]] = []
    current_sha = ""
    current_subject = ""
    for line in log_proc.stdout.splitlines():
        if line == "--FILES--":
            entries.append((current_sha + "|" + current_subject, []))
            continue
        if "|" in line and len(line.split("|")[0]) == 40:
            current_sha, current_subject = line.split("|", 1)
        elif not entries:
            current_subject = line  # first subject line
        elif entries:
            # Inside a commit's files section — collect via separate git call
            pass

    # For simplicity, do a per-commit file lookup
    out: list[tuple[str, str, list[str]]] = []
    for entry, _ in entries:
        sha, subject = entry.split("|", 1)
        diff_proc = subprocess.run(
            ["git", "diff-tree", "--no-commit-id", "--name-only", "-r", sha],
            capture_output=True,
            text=True,
            check=False,
            cwd=_ROOT,
        )
        files = [line.strip() for line in diff_proc.stdout.splitlines() if line.strip()]
        out.append((sha, subject, files))
    return out


def emit_markdown(rows: list[Classification]) -> str:
    """Render the classification table as markdown."""
    lines = [
        "# main-classification",
        "",
        "Auto-generated by `scripts/port_classify.py` against "
        f"`{_MERGE_BASE}..{_ORIGIN}`. Re-run the script to refresh.",
        "",
        "| sha | subject | cluster | lane | notes |",
        "|-----|---------|---------|------|-------|",
    ]
    for r in rows:
        sha_short = r.sha[:7]
        # Escape pipe characters in subject and notes
        subject = r.subject.replace("|", "\\|")
        notes = r.notes.replace("|", "\\|")
        lines.append(f"| {sha_short} | {subject} | {r.cluster} | {r.lane} | {notes} |")
    lines.append("")
    return "\n".join(lines)


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--base", default=_MERGE_BASE)
    parser.add_argument("--head", default=_ORIGIN)
    parser.add_argument("--output", default="docs/port/main-classification.md")
    args = parser.parse_args()

    rows = [classify_commit(sha, subject, files) for sha, subject, files in _git_log(args.base, args.head)]
    md = emit_markdown(rows)
    out_path = _ROOT / args.output
    out_path.parent.mkdir(parents=True, exist_ok=True)
    out_path.write_text(md, encoding="utf-8")
    print(f"Wrote {len(rows)} rows to {out_path}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
def _git_diff_stats(paths: list[str]) -> tuple[int, list[str]]:
    """Return (commit count, top-5 changed files) for the cluster paths."""
    log = subprocess.run(
        ["git", "log", f"{_MERGE_BASE}..{_ORIGIN}", "--oneline", "--", *paths],
        capture_output=True,
        text=True,
        check=False,
        cwd=_ROOT,
    )
    lines = [l for l in log.stdout.splitlines() if l]
    n = len(lines)
    diff = subprocess.run(
        ["git", "diff", f"{_MERGE_BASE}..{_ORIGIN}", "--name-only", "--", *paths],
        capture_output=True,
        text=True,
        check=False,
        cwd=_ROOT,
    )

    return n, files
```

```python
"""Apply one cluster's end-state delta as a structured commit.

def _diff_patch(paths: list[str]) -> str:
    p = subprocess.run(
        ["git", "diff", f"{_MERGE_BASE}..{_ORIGIN}", "--", *paths],
        capture_output=True,
        text=True,
        check=False,
        cwd=_ROOT,
    )
    return p.stdout

    p = subprocess.run(
        ["git", "apply", "--check"],

    parser.add_argument("--cluster-id", required=True)
    parser.add_argument("--lane", choices=["A", "B", "C"], required=True)
    parser.add_argument("--name", required=True)
    parser.add_argument("--lock-impact", default="none")
    parser.add_argument("--test-plan", default="no coverage")
    parser.add_argument("--source-range", default="bae32d8c..origin/main")
    parser.add_argument("--apply", action="store_true",
                        help="Apply the patch (default: check only)")
    parser.add_argument("--patch-out", default="/tmp/cluster-patch.diff")
    parser.add_argument("--msg-out", default="/tmp/cluster-msg.txt")
    args = parser.parse_args()

    patch = _diff_patch(args.paths)
    Path(args.patch_out).write_text(patch, encoding="utf-8")
    ok, err = _apply_check(patch)
    if not ok:
        print(f"git apply --check failed:\n{err}", file=sys.stderr)
        return 1

    msg = (
        f"port(cluster-{args.cluster_id}): {args.name}\n\n"
        f"Source: {args.source_range}.\n"
        f"Lane: {args.lane}.\n"
        f"Lock impact: {args.lock_impact}.\n"
        f"Test plan: {args.test_plan}.\n"
    )
    Path(args.msg_out).write_text(msg, encoding="utf-8")
    print(f"Patch: {args.patch_out} ({len(patch.splitlines())} lines)")
    print(f"Commit-msg: {args.msg_out}")
    print(msg)

    if args.apply:
        apply_proc = subprocess.run(
            ["git", "apply"],
            input=patch,
            capture_output=True,
            text=True,
            check=False,
            cwd=_ROOT,
        )
        if apply_proc.returncode != 0:
            print(f"git apply failed: {apply_proc.stderr}", file=sys.stderr)
            return 1
        print(f"Applied. Now: run gates, then `git commit -F {args.msg_out}`.")

    return 0


if __name__ == "__main__":
    sys.exit(main())
```
